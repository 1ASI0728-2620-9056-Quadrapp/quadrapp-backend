BEGIN;

-- ============================================================
-- 1. IAM - IDENTITY AND ACCESS MANAGEMENT
-- ============================================================

-- ------------------------------------------------------------
-- 1.1 Institutions
-- ------------------------------------------------------------

CREATE TABLE public.institutions (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name varchar(150) NOT NULL UNIQUE,
    active boolean NOT NULL DEFAULT true,
    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now()
);

-- ------------------------------------------------------------
-- 1.2 Users
-- ------------------------------------------------------------

CREATE TABLE public.users (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    institution_id bigint REFERENCES public.institutions(id),
    invited_by bigint REFERENCES public.users(id),

    email varchar(254) NOT NULL UNIQUE,
    full_name varchar(120),

    role varchar(15) NOT NULL
        CHECK (role IN (
            'DRIVER',
            'OPERATOR',
            'ADMIN',
            'PLATFORM_ADMIN'
        )),

    status varchar(10) NOT NULL DEFAULT 'ACTIVE'
        CHECK (status IN (
            'INVITED',
            'ACTIVE',
            'DISABLED'
        )),

    invitation_token_hash varchar(255) UNIQUE,
    invitation_expires_at timestamptz,

    verified_at timestamptz,
    last_login_at timestamptz,

    locale varchar(6) NOT NULL DEFAULT 'es-419'
        CHECK (locale IN ('es-419', 'en-US')),

    push_enabled boolean NOT NULL DEFAULT true,
    saturation_alerts boolean NOT NULL DEFAULT true,
    low_arrival_alerts boolean NOT NULL DEFAULT true,
    stale_data_alerts boolean NOT NULL DEFAULT true,
    event_alerts boolean NOT NULL DEFAULT false,

    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT ck_users_lowercase_email
        CHECK (email = lower(email)),

    CONSTRAINT ck_users_invited
        CHECK (
            status <> 'INVITED'
            OR (
                invitation_token_hash IS NOT NULL
                AND invitation_expires_at IS NOT NULL
            )
        ),

    CONSTRAINT ck_users_platform_admin
        CHECK (
            (role = 'PLATFORM_ADMIN') =
            (institution_id IS NULL)
        )
);

-- ------------------------------------------------------------
-- 1.3 Institution Domains
-- ------------------------------------------------------------

CREATE TABLE public.institution_domains (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    institution_id bigint NOT NULL
        REFERENCES public.institutions(id),

    verified_by bigint REFERENCES public.users(id),

    domain varchar(120) NOT NULL UNIQUE,
    verified_at timestamptz,
    created_at timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT ck_institution_domains_lowercase
        CHECK (domain = lower(domain))
);

-- ------------------------------------------------------------
-- 1.4 Refresh Tokens
-- ------------------------------------------------------------

CREATE TABLE public.refresh_tokens (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    user_id bigint NOT NULL
        REFERENCES public.users(id),

    token_hash varchar(255) NOT NULL UNIQUE,
    expires_at timestamptz NOT NULL,
    revoked_at timestamptz,
    created_at timestamptz NOT NULL DEFAULT now()
);

-- ============================================================
-- 2. PARKING CONFIGURATION
-- ============================================================

-- ------------------------------------------------------------
-- 2.1 Parking Lots
-- ------------------------------------------------------------

CREATE TABLE public.parking_lots (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    institution_id bigint NOT NULL
        REFERENCES public.institutions(id),

    created_by bigint REFERENCES public.users(id),

    name varchar(120) NOT NULL,
    address varchar(255),

    latitude numeric(9,6)
        CHECK (latitude BETWEEN -90 AND 90),

    longitude numeric(9,6)
        CHECK (longitude BETWEEN -180 AND 180),

    opens_at time,
    closes_at time,

    saturation_threshold_pct smallint NOT NULL DEFAULT 90
        CHECK (saturation_threshold_pct BETWEEN 1 AND 100),

    low_battery_threshold_pct smallint NOT NULL DEFAULT 15
        CHECK (low_battery_threshold_pct BETWEEN 1 AND 100),

    stale_after_seconds integer NOT NULL DEFAULT 180
        CHECK (stale_after_seconds > 0),

    flow_window_minutes smallint NOT NULL DEFAULT 15
        CHECK (flow_window_minutes > 0),

    accuracy_margin_pct smallint NOT NULL DEFAULT 10
        CHECK (accuracy_margin_pct BETWEEN 1 AND 100),

    occupied_debounce_s smallint NOT NULL DEFAULT 10
        CHECK (occupied_debounce_s > 0),

    freed_debounce_s smallint NOT NULL DEFAULT 20
        CHECK (freed_debounce_s > 0),

    layout_version integer NOT NULL DEFAULT 0,
    layout_published_at timestamptz,

    active boolean NOT NULL DEFAULT true,

    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT uk_parking_lots_institution_name
        UNIQUE (institution_id, name),

    CONSTRAINT ck_parking_lots_open_pair
        CHECK (
            (opens_at IS NULL) =
            (closes_at IS NULL)
        )
);

-- ------------------------------------------------------------
-- 2.2 Zones
-- ------------------------------------------------------------

CREATE TABLE public.zones (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    lot_id bigint NOT NULL
        REFERENCES public.parking_lots(id),

    name varchar(60) NOT NULL,
    description varchar(80),

    sort_order smallint NOT NULL DEFAULT 0,

    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT uk_zones_lot_name
        UNIQUE (lot_id, name)
);

-- ------------------------------------------------------------
-- 2.3 Levels
-- ------------------------------------------------------------

CREATE TABLE public.levels (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    zone_id bigint NOT NULL
        REFERENCES public.zones(id),

    name varchar(40) NOT NULL,
    sort_order smallint NOT NULL DEFAULT 0,

    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT uk_levels_zone_name
        UNIQUE (zone_id, name)
);

-- ------------------------------------------------------------
-- 2.4 Spaces
-- ------------------------------------------------------------

CREATE TABLE public.spaces (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    level_id bigint NOT NULL
        REFERENCES public.levels(id),

    label varchar(20) NOT NULL,

    type varchar(10) NOT NULL DEFAULT 'STANDARD'
        CHECK (type IN (
            'STANDARD',
            'ACCESSIBLE'
        )),

    pos_x numeric(7,2),
    pos_y numeric(7,2),

    status varchar(10) NOT NULL DEFAULT 'UNKNOWN'
        CHECK (status IN (
            'FREE',
            'OCCUPIED',
            'UNKNOWN'
        )),

    unknown_reason varchar(25) DEFAULT 'NO_SENSOR'
        CHECK (
            unknown_reason IS NULL
            OR unknown_reason IN (
                'NO_HEARTBEAT',
                'LOW_BATTERY',
                'NO_SENSOR'
            )
        ),

    status_since timestamptz NOT NULL DEFAULT now(),
    last_event_at timestamptz,

    active boolean NOT NULL DEFAULT true,

    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT uk_spaces_level_label
        UNIQUE (level_id, label),

    CONSTRAINT ck_spaces_unknown_reason
        CHECK (
            (status = 'UNKNOWN') =
            (unknown_reason IS NOT NULL)
        )
);

-- ------------------------------------------------------------
-- 2.5 Accesses
-- ------------------------------------------------------------

CREATE TABLE public.accesses (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    lot_id bigint NOT NULL
        REFERENCES public.parking_lots(id),

    name varchar(60) NOT NULL,

    type varchar(15) NOT NULL
        CHECK (type IN (
            'ENTRY',
            'EXIT',
            'BIDIRECTIONAL'
        )),

    flow_counting_enabled boolean NOT NULL DEFAULT false,

    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT uk_accesses_lot_name
        UNIQUE (lot_id, name)
);

-- ------------------------------------------------------------
-- 2.6 Calendar Events
-- ------------------------------------------------------------

CREATE TABLE public.calendar_events (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    institution_id bigint NOT NULL
        REFERENCES public.institutions(id),

    lot_id bigint REFERENCES public.parking_lots(id),
    zone_id bigint REFERENCES public.zones(id),
    created_by bigint REFERENCES public.users(id),

    type varchar(15) NOT NULL
        CHECK (type IN (
            'CLASSES',
            'EXAMS',
            'RECESS',
            'CAMPUS_EVENT'
        )),

    name varchar(120) NOT NULL,

    impact varchar(6) NOT NULL DEFAULT 'MEDIUM'
        CHECK (impact IN (
            'LOW',
            'MEDIUM',
            'HIGH'
        )),

    starts_at timestamptz NOT NULL,
    ends_at timestamptz NOT NULL,

    cancelled_at timestamptz,

    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT ck_events_window
        CHECK (ends_at > starts_at),

    CONSTRAINT ck_events_scope
        CHECK (
            zone_id IS NULL
            OR lot_id IS NOT NULL
        )
);

-- ============================================================
-- 3. PARKING SENSING
-- ============================================================

-- ------------------------------------------------------------
-- 3.1 Devices
-- ------------------------------------------------------------

CREATE TABLE public.devices (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    lot_id bigint NOT NULL
        REFERENCES public.parking_lots(id),

    gateway_id bigint REFERENCES public.devices(id),
    space_id bigint REFERENCES public.spaces(id),
    access_id bigint REFERENCES public.accesses(id),

    replaced_by bigint REFERENCES public.devices(id),
    registered_by bigint REFERENCES public.users(id),

    hardware_id varchar(80) NOT NULL UNIQUE,
    api_key_hash varchar(255),

    type varchar(20) NOT NULL
        CHECK (type IN (
            'SPACE_SENSOR',
            'PASS_SENSOR',
            'GATEWAY'
        )),

    directional boolean,

    expected_report_interval_s integer NOT NULL DEFAULT 60
        CHECK (expected_report_interval_s > 0),

    status varchar(15) NOT NULL DEFAULT 'OPERATIONAL'
        CHECK (status IN (
            'OPERATIONAL',
            'UNREACHABLE',
            'LOW_BATTERY',
            'DECOMMISSIONED'
        )),

    battery_pct smallint
        CHECK (battery_pct BETWEEN 0 AND 100),

    firmware_version varchar(30),

    last_seen_at timestamptz,
    decommissioned_at timestamptz,

    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT ck_devices_gateway_assignment
        CHECK (
            type <> 'GATEWAY'
            OR (
                space_id IS NULL
                AND access_id IS NULL
            )
        ),

    CONSTRAINT ck_devices_space_assignment
        CHECK (
            type <> 'SPACE_SENSOR'
            OR access_id IS NULL
        ),

    CONSTRAINT ck_devices_pass_assignment
        CHECK (
            type <> 'PASS_SENSOR'
            OR (
                space_id IS NULL
                AND directional IS NOT NULL
            )
        )
);

-- ------------------------------------------------------------
-- 3.2 Device Incidents
-- ------------------------------------------------------------

CREATE TABLE public.device_incidents (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    lot_id bigint NOT NULL
        REFERENCES public.parking_lots(id),

    device_id bigint REFERENCES public.devices(id),
    reported_by bigint REFERENCES public.users(id),
    resolved_by bigint REFERENCES public.users(id),

    hardware_id varchar(80),

    type varchar(25) NOT NULL
        CHECK (type IN (
            'NO_HEARTBEAT',
            'LOW_BATTERY',
            'UNREGISTERED_DEVICE',
            'UNMAPPED_SENSOR',
            'INVALID_MESSAGE',
            'USER_REPORT'
        )),

    category varchar(20)
        CHECK (category IN (
            'NO_COMMUNICATION',
            'WRONG_DATA',
            'LOW_BATTERY',
            'OTHER'
        )),

    detail text,

    detected_at timestamptz NOT NULL DEFAULT now(),
    resolved_at timestamptz,

    CONSTRAINT ck_incidents_user_report
        CHECK (
            (type = 'USER_REPORT') =
            (
                reported_by IS NOT NULL
                AND category IS NOT NULL
            )
        )
);

-- ------------------------------------------------------------
-- 4.1 Sensor Events
-- ------------------------------------------------------------

CREATE TABLE public.sensor_events (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    event_id varchar(100) NOT NULL UNIQUE,

    device_id bigint NOT NULL
        REFERENCES public.devices(id),

    space_id bigint REFERENCES public.spaces(id),
    access_id bigint REFERENCES public.accesses(id),

    event_type varchar(10) NOT NULL
        CHECK (event_type IN (
            'OCCUPIED',
            'FREED',
            'ENTRY',
            'EXIT'
        )),

    occurred_at timestamptz NOT NULL,
    received_at timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT ck_sensor_events_kind
        CHECK (
            (
                event_type IN ('OCCUPIED', 'FREED')
                AND space_id IS NOT NULL
                AND access_id IS NULL
            )
            OR
            (
                event_type IN ('ENTRY', 'EXIT')
                AND access_id IS NOT NULL
                AND space_id IS NULL
            )
        )
);

-- ------------------------------------------------------------
-- 4.2 Occupancy Snapshots
-- ------------------------------------------------------------

CREATE TABLE public.occupancy_snapshots (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    zone_id bigint NOT NULL
        REFERENCES public.zones(id),

    taken_at timestamptz NOT NULL,

    total_spaces smallint NOT NULL,
    free_count smallint NOT NULL,
    occupied_count smallint NOT NULL,
    unknown_count smallint NOT NULL,

    CONSTRAINT uk_occupancy_snapshots_zone_taken
        UNIQUE (zone_id, taken_at),

    CONSTRAINT ck_occupancy_snapshots_total
        CHECK (
            free_count
            + occupied_count
            + unknown_count
            = total_spaces
        )
);

-- 5. PREDICTION & ADVISORY

-- ------------------------------------------------------------
-- 5.1 Forecasts
-- ------------------------------------------------------------

CREATE TABLE public.forecasts (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    lot_id bigint NOT NULL
        REFERENCES public.parking_lots(id),

    horizon_min smallint NOT NULL
        CHECK (horizon_min IN (
            15, 30, 45, 60
        )),

    generated_at timestamptz NOT NULL DEFAULT now(),
    target_at timestamptz NOT NULL,

    expected_occupancy_pct numeric(5,2) NOT NULL
        CHECK (
            expected_occupancy_pct BETWEEN 0 AND 100
        ),

    confidence numeric(4,3) NOT NULL
        CHECK (confidence BETWEEN 0 AND 1),

    confidence_level varchar(6) NOT NULL
        CHECK (confidence_level IN (
            'HIGH',
            'MEDIUM',
            'LOW'
        )),

    method varchar(15) NOT NULL
        CHECK (method IN (
            'PRIMARY_MODEL',
            'FALLBACK'
        )),

    model_version varchar(40) NOT NULL,

    saturation_expected_at timestamptz,

    observed_occupancy_pct numeric(5,2)
        CHECK (
            observed_occupancy_pct BETWEEN 0 AND 100
        ),

    abs_error numeric(5,2),

    CONSTRAINT uk_forecasts_lot_horizon_generated
        UNIQUE (
            lot_id,
            horizon_min,
            generated_at
        ),

    CONSTRAINT ck_forecasts_fallback_confidence
        CHECK (
            method <> 'FALLBACK'
            OR confidence_level = 'LOW'
        )
);

-- ------------------------------------------------------------
-- 6.1 Alert Subscriptions
-- ------------------------------------------------------------

CREATE TABLE public.alert_subscriptions (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    user_id bigint NOT NULL
        REFERENCES public.users(id),

    lot_id bigint NOT NULL
        REFERENCES public.parking_lots(id),

    days smallint[] NOT NULL
        CHECK (
            cardinality(days) BETWEEN 1 AND 7
            AND days <@ ARRAY[1,2,3,4,5,6,7]::smallint[]
        ),

    slot_from time NOT NULL,
    slot_to time NOT NULL,

    active boolean NOT NULL DEFAULT true,

    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT ck_alert_subscription_window
        CHECK (slot_to > slot_from)
);

-- ------------------------------------------------------------
-- 6.2 Push Tokens
-- ------------------------------------------------------------

CREATE TABLE public.push_tokens (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    user_id bigint NOT NULL
        REFERENCES public.users(id),

    token varchar(512) NOT NULL UNIQUE,

    platform varchar(10) NOT NULL
        CHECK (platform IN (
            'ANDROID',
            'IOS',
            'WEB'
        )),

    valid boolean NOT NULL DEFAULT true,

    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now()
);

-- ------------------------------------------------------------
-- 6.3 Notifications
-- ------------------------------------------------------------

CREATE TABLE public.notifications (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    user_id bigint NOT NULL
        REFERENCES public.users(id),

    lot_id bigint NOT NULL
        REFERENCES public.parking_lots(id),

    subscription_id bigint
        REFERENCES public.alert_subscriptions(id),

    forecast_id bigint
        REFERENCES public.forecasts(id),

    type varchar(15) NOT NULL
        CHECK (type IN (
            'SATURATION',
            'LOW_ARRIVAL'
        )),

    condition_key varchar(150) NOT NULL,

    status varchar(10) NOT NULL
        CHECK (status IN (
            'SENT',
            'SKIPPED',
            'FAILED'
        )),

    created_at timestamptz NOT NULL DEFAULT now(),

    CONSTRAINT uk_notifications_user_condition
        UNIQUE (user_id, condition_key)
);
